1 · Read the camera program

One per pair · SDK 12.0.0 · Circle setup, Start, loop, pose check, update, cleanup.

 1  package org.firstinspires.ftc.teamcode;
 2  
 3  import com.qualcomm.robotcore.eventloop.opmode.LinearOpMode;
 4  import com.qualcomm.robotcore.eventloop.opmode.TeleOp;
 5  import org.firstinspires.ftc.robotcore.external.hardware.camera.WebcamName;
 6  import org.firstinspires.ftc.vision.VisionPortal;
 7  import org.firstinspires.ftc.vision.apriltag.AprilTagDetection;
 8  import org.firstinspires.ftc.vision.apriltag.AprilTagProcessor;
 9  import java.util.List;
10  
11  @TeleOp(name = "M3 AprilTag Telemetry", group = "BIOBUZZ")
12  public class Meeting3AprilTagTelemetry extends LinearOpMode {
13      @Override
14      public void runOpMode() throws InterruptedException {
15          // SETUP: the configured webcam name must match exactly.
16          AprilTagProcessor tags = AprilTagProcessor.easyCreateWithDefaults();
17          VisionPortal camera = VisionPortal.easyCreateWithDefaults(
18                  hardwareMap.get(WebcamName.class, "Webcam 1"), tags);
19  
20          // The finally block releases the camera even if we Stop during INIT.
21          try {
22              telemetry.addLine("Camera opening. Press Start when ready.");
23              telemetry.update();
24              waitForStart();
25  
26              while (opModeIsActive()) {
27                  List<AprilTagDetection> detections = tags.getDetections();
28                  telemetry.addData("Camera", camera.getCameraState());
29                  telemetry.addData("Detections", detections.size());
30                  if (detections.isEmpty()) {
31                      telemetry.addLine("No tags visible yet.");
32                  }
33  
34                  // READ: single tags and clusters share the same pose fields.
35                  for (AprilTagDetection detection : detections) {
36                      if (detection.ftcPose != null) {
37                          // SHOW: default units are inches and degrees.
38                          telemetry.addLine("--- Target pose ---");
39                          // Range: camera-to-target distance in the X-Y plane (inches).
40                          // It ignores height (Z): sqrt(X*X + Y*Y).
41                          telemetry.addData("Range (in)", "%.1f", detection.ftcPose.range);
42                          // Bearing: left/right angle (degrees); left +, right -, centered 0.
43                          telemetry.addData("Bearing (deg)", "%.1f", detection.ftcPose.bearing);
44                          // Elevation: up/down angle (degrees); above +, below -, level 0.
45                          telemetry.addData("Elevation (deg)", "%.1f", detection.ftcPose.elevation);
46                          telemetry.addData("X / Y / Z (in)", "%.1f / %.1f / %.1f",
47                                  detection.ftcPose.x, detection.ftcPose.y, detection.ftcPose.z);
48                      } else {
49                          telemetry.addLine("Tag seen; no distance. Check tag library/size.");
50                      }
51                  }
52                  telemetry.update();
53                  sleep(50);
54              }
55          } finally {
56              camera.close();
57          }
58      }
59  }
